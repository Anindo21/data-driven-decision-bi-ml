-- =====================================
-- Dimension: Customer
-- =====================================
CREATE TABLE dbo.DimCustomer (
    CustomerId        INT PRIMARY KEY,
    CustomerSegment   NVARCHAR(50),
    CustomerCountry   NVARCHAR(50),
    CustomerState     NVARCHAR(50),
    CustomerCity      NVARCHAR(100),
    CustomerStreet    NVARCHAR(100),
    CustomerZipcode   NVARCHAR(20)
);
-- =====================================
-- Dimension: Product
-- =====================================
CREATE TABLE dbo.DimProduct (
    ProductCardId     INT PRIMARY KEY,      
    ProductCategoryId INT,
    CategoryName      NVARCHAR(100),
    ProductName       NVARCHAR(100),
    ProductPrice      DECIMAL(18,2),
    ProductStatus     BIT
);
-- =====================================
-- Dimension: Department
-- =====================================
CREATE TABLE dbo.DimDepartment (
    DepartmentId      INT IDENTITY(1,1) PRIMARY KEY,
    DepartmentName    NVARCHAR(100),
    Market            NVARCHAR(100)
);
-- =====================================
-- Dimension: Location
-- =====================================
CREATE TABLE dbo.DimLocation (
    LocationId        INT IDENTITY(1,1) PRIMARY KEY,
    OrderRegion       NVARCHAR(100),
    OrderCity         NVARCHAR(100),
    OrderState        NVARCHAR(100),
    OrderCountry      NVARCHAR(100)
);
-- =====================================
-- Dimension: Shipping
-- =====================================
CREATE TABLE dbo.DimShipping (
    ShippingId              INT IDENTITY(1,1) PRIMARY KEY,
    ShippingMode            NVARCHAR(50),
    DaysForShippingReal     INT,
    DaysForShipmentScheduled INT
);
-- =====================================
-- Dimension: Order Status
-- =====================================
CREATE TABLE dbo.DimOrderStatus (
    OrderStatusId   INT IDENTITY(1,1) PRIMARY KEY,
    OrderStatus     NVARCHAR(50)
);

-- =====================================
-- Dimension: Transaction Type
-- =====================================
CREATE TABLE dbo.DimTransactionType (
    TypeId  INT IDENTITY(1,1) PRIMARY KEY,
    Type               NVARCHAR(20)
);
-- =====================================
-- Dimension: Date
-- =====================================
CREATE TABLE dbo.DimDate (
    DateId         DATE PRIMARY KEY,
    Year           INT,
    Quarter        INT,
    Month          INT,
    MonthName      NVARCHAR(20),
    Day            INT,
    DayOfWeek      INT,
    DayName        NVARCHAR(20)
);

-- =============================================
-- Fact Table: FactOrderItem
-- =============================================
CREATE TABLE dbo.FactOrderItem (
    FactOrderItemId        INT IDENTITY(1,1) PRIMARY KEY,
    OrderId                INT,
    OrderItemId            INT,
    CustomerId             INT,    -- FK to DimCustomer
    ProductCardId          INT,    -- FK to DimProduct (business key, not surrogate)
    DepartmentId           INT,    -- FK to DimDepartment (surrogate key)
    LocationId             INT,    -- FK to DimLocation (surrogate key)
    ShippingId             INT,    -- FK to DimShipping (surrogate key)
    OrderStatusId          INT,    -- FK to DimOrderStatus (surrogate key)
    TypeId                 INT,    -- FK to DimTransactionType (surrogate key)
    OrderDateId            DATE,   -- FK to DimDate (date PK)
    ShippingDateId         DATE,   -- FK to DimDate (date PK)
    OrderItemQuantity      INT,
    OrderItemDiscount      DECIMAL(18,2),
    OrderItemDiscountRate  DECIMAL(10,6),
    OrderItemProductPrice  DECIMAL(18,2),
    OrderItemProfitRatio   DECIMAL(10,6),
    Sales                  DECIMAL(18,2),
    OrderItemTotal         DECIMAL(18,2),
    Benefitperorder        DECIMAL(18,2),
    OrderProfitPerOrder    DECIMAL(18,2),
    DeliveryStatus         NVARCHAR(50),
    Late_delivery_risk     BIT,

    -- Foreign Key Constraints
    FOREIGN KEY (CustomerId)        REFERENCES dbo.DimCustomer(CustomerId),
    FOREIGN KEY (ProductCardId)     REFERENCES dbo.DimProduct(ProductCardId),
    FOREIGN KEY (DepartmentId)      REFERENCES dbo.DimDepartment(DepartmentId),
    FOREIGN KEY (LocationId)        REFERENCES dbo.DimLocation(LocationId),
    FOREIGN KEY (ShippingId)        REFERENCES dbo.DimShipping(ShippingId),
    FOREIGN KEY (OrderStatusId)     REFERENCES dbo.DimOrderStatus(OrderStatusId),
    FOREIGN KEY (TypeId)            REFERENCES dbo.DimTransactionType(TypeId),
    FOREIGN KEY (OrderDateId)       REFERENCES dbo.DimDate(DateId),
    FOREIGN KEY (ShippingDateId)    REFERENCES dbo.DimDate(DateId)
);


-- Indexes for performance 
CREATE NONCLUSTERED INDEX idx_FactOrderItem_DepartmentId ON dbo.FactOrderItem(DepartmentId);
CREATE NONCLUSTERED INDEX idx_FactOrderItem_ProductCardId ON dbo.FactOrderItem(ProductCardId);
CREATE NONCLUSTERED INDEX idx_FactOrderItem_CustomerId ON dbo.FactOrderItem(CustomerId);
CREATE NONCLUSTERED INDEX idx_FactOrderItem_LocationId ON dbo.FactOrderItem(LocationId);
CREATE NONCLUSTERED INDEX idx_FactOrderItem_ShippingId ON dbo.FactOrderItem(ShippingId);
CREATE NONCLUSTERED INDEX idx_FactOrderItem_OrderStatusId ON dbo.FactOrderItem(OrderStatusId);
CREATE NONCLUSTERED INDEX idx_FactOrderItem_TypeId ON dbo.FactOrderItem(TypeId);
CREATE NONCLUSTERED INDEX idx_FactOrderItem_OrderDateId ON dbo.FactOrderItem(OrderDateId);
CREATE NONCLUSTERED INDEX idx_FactOrderItem_ShippingDateId ON dbo.FactOrderItem(ShippingDateId);
